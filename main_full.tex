% ============================================================
%  目录条目样式
%   · 编号占 5em 固定列；页码统一右对齐到右边界（\parfillskip=0pt）
%   · 姓名与页码之间不加引导符、不留空白（页码紧贴姓名）
%   · 版式由宏内 \settowidth 实测宽度自动选择（不依赖字符估算）：
%       A = 编号+标题+最小引导符+姓名+页码(占位)   B = 编号+标题+最小引导符
%       ① A < 行宽            ⇒ 单行：标题…引导符…姓名（页码紧贴）
%       ② A ≥ 行宽 且 B < 行宽 ⇒ 标题占一行（引导符到右边界），姓名另起
%                                一行，姓名前有引导符，页码右对齐
%       ③ B ≥ 行宽（标题自身折行）⇒ 自然流动，姓名接在标题末行，
%                                折行处悬挂缩进 5em
%   · 引导符胶水分级：姓名前用 fill（高阶）→ 同行时吸收剩余空间；
%     姓名后用 fil（低阶）→ 仅姓名折行时生效，把页码推到右边界
%   · 页码宽度用 \hphantom{000} 占位测量，避免首轮编译时 \pageref 未定
%     导致版式在两个数值间反复横跳
%   · 条目内部不允许断页（club/widow/interline penalty = 10000）
% ============================================================
\newlength{\tocwdA}
\newlength{\tocwdB}

\newcommand{\tocentry}[4]{%
  \begingroup
  \emergencystretch=3em
  \clubpenalty=10000
  \widowpenalty=10000
  \interlinepenalty=10000
  \setlength{\parindent}{0pt}%
  \setlength{\parfillskip}{0pt}%
  \settowidth{\tocwdA}{\wuhao\makebox[5em][l]{#1}#3\hskip .5em{\kaishu\wuhao #4}\wuhao（\hphantom{000}）}%
  \settowidth{\tocwdB}{\wuhao\makebox[5em][l]{#1}#3\hskip .5em}%
  \ifdim\tocwdA<\linewidth
    % —— ① 单行放得下 ——
    \hangindent=5em
    \hangafter=1
    \noindent\makebox[5em][l]{\wuhao #1}%
    {\wuhao\hyperlink{paper:#2}{#3}}%
    \nobreak\leaders\hbox to .5em{\hss.\hss}\hfill\nobreak
    {\kaishu\wuhao #4}%
    {\wuhao（\pageref{paper:#2}）}%
  \else\ifdim\tocwdB<\linewidth
    % —— ② 标题占一行，姓名另起一行 ——
    \hangindent=5em
    \hangafter=1
    \noindent\makebox[5em][l]{\wuhao #1}%
    {\wuhao\hyperlink{paper:#2}{#3}}%
    \nobreak\leaders\hbox to .5em{\hss.\hss}\hfill\nobreak
    \mbox{}%
    \par\nobreak\vspace{3pt}%
    \hangindent=6.5em
    \hangafter=1
    \noindent\hspace*{5em}%
    \nobreak\leaders\hbox to .5em{\hss.\hss}\hskip 1.5em\relax
    \nobreak\leaders\hbox to .5em{\hss.\hss}\hskip 0pt plus 1fill\relax
    \nobreak{\kaishu\wuhao #4}\nobreak
    \leaders\hbox to .5em{\hss.\hss}\hskip 0pt plus 1fil\relax
    {\wuhao（\pageref{paper:#2}）}%
  \else
    % —— ③ 标题自身折行：自然流动，姓名接在标题末行 ——
    \hangindent=5em
    \hangafter=1
    \noindent\makebox[5em][l]{\wuhao #1}%
    {\wuhao\hyperlink{paper:#2}{#3}}%
    \nobreak\leaders\hbox to .5em{\hss.\hss}\hfill\nobreak
    {\kaishu\wuhao #4}%
    {\wuhao（\pageref{paper:#2}）}%
  \fi\fi
  \par\vspace{5pt}%
  \endgroup
}
